\begin{frame}{From Chatbots to Agents}
    \begin{columns}[c]
        \begin{column}{0.52\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.7\textheight,keepaspectratio]{images/agentic-ai/agentic_agent_loop_memory_tools_goals.png}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \setlength\itemsep{0.7em}
                \item A \textbf{chatbot} answers your question.
                \item An \textbf{agent} is given a goal and takes steps to reach it.
                \item It can use \textbf{tools}: search the web, read files, send an email, run a program.
                \item It looks at the result of each step and decides what to do next.
            \end{itemize}

            \vspace{0.5em}
            \small \textit{Example: ``Find three hotels near the venue under my budget and put them in a table.''}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{AI for Science: AlphaFold}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.68\textheight,keepaspectratio]{images/science/alphafold-3-protein.png}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \setlength\itemsep{0.7em}
                \item Proteins are the tiny machines of life. Their \textbf{shape} decides what they do.
                \item Finding one protein's shape in a laboratory could take years.
                \item \textbf{AlphaFold} predicts the shape with AI, in minutes.
                \item It has predicted more than 200 million protein shapes, shared freely with scientists.
                \item Its creators shared the \textbf{2024 Nobel Prize in Chemistry}.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
