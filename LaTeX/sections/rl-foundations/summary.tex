\begin{frame}{}
    \LARGE Reinforcement Learning: \textbf{Summary}
\end{frame}

\begin{frame}{Summary}
    \begin{itemize}
        \item Reinforcement Learning (RL) is a powerful paradigm for \textbf{sequential decision-making}.
        \item Markov Decision Processes (MDPs) provide a \textbf{mathematical foundation}.
        \item Q-learning enables \textbf{model-free control}.
        \item Despite its power, RL has \textbf{significant limitations}.
        \item Research is ongoing to make RL \textbf{scalable, robust, and safe}.
    \end{itemize}
\end{frame}

    \hline

    \hline
    DQN & Q-learning + parameterized $Q_\theta$ + Bellman error \\

    DDPG & Q-learning extended to continuous actions \\
    SAC & Bellman equation + entropy term added to reward \\
    Exploration & $\varepsilon$-greedy extended + bandit formalisation \\
    Model-Based & MDP transition model $P(s'|s,a)$ made explicit \\
    RLHF & Policy $\pi_\theta$ + reward design + partial observability \\
    Frontiers & All of the above \\
    \hline
    \end{tabular}
\end{frame}
