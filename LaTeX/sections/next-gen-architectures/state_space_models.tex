\subsection{State-Space Models}

\begin{frame}{A Second Road to the Same Place: State-Space Models}
    \begin{itemize}\itemsep0.5em
        \item Linear attention started from attention and removed the cache. State-space models (SSMs) started from \textbf{control theory} and arrived at the same recurrence.
        \item Discretise a linear system $h'(t)=Ah(t)+Bx(t),\; y(t)=Ch(t)$:
        \[
            h_t=\bar{A}\,h_{t-1}+\bar{B}\,x_t, \qquad y_t = C\,h_t
        \]
        \item \textbf{S4} (2021): fixed $A,B,C$ make the whole map one long convolution, so training is parallel. It solved long-range benchmarks that Transformers failed, such as Path-X at length 16k.
        \item The catch is \emph{time-invariance}. One kernel treats every token alike. The model cannot keep or drop a token because of \textbf{what it is}.
    \end{itemize}
    \src{Gu et al., ``Efficiently Modeling Long Sequences with Structured State Spaces,'' \href{https://arxiv.org/abs/2111.00396}{arXiv:2111.00396}.}
\end{frame}

\begin{frame}{Mamba: Let the Input Choose What to Remember}
    \begin{itemize}\itemsep0.35em
        \item \textbf{One change}: make $B$, $C$ and the step size $\Delta$ functions of the current token.
        \item Read $\Delta_t$ as a gate. Large: reset the state toward this token. Small: ignore it and hold.
        \item Selectivity breaks the convolution, so Mamba runs a \textbf{hardware-aware parallel scan} instead.
        \item Payoff reported in the paper: about 5$\times$ the generation throughput of a same-size Transformer.
    \end{itemize}
    \centering
    \includegraphics[width=0.86\linewidth,height=0.34\textheight,keepaspectratio]{images/next-gen-architectures/mamba_paper_fig1_selective_ssm_overview.png}
    \src{Gu and Dao, ``Mamba: Linear-Time Sequence Modeling with Selective State Spaces,'' \href{https://arxiv.org/abs/2312.00752}{arXiv:2312.00752}, Figure 1.}
\end{frame}

\begin{frame}{Mamba-2: The Two Roads Meet}
    \begin{columns}[T,onlytextwidth]
        \column{0.62\textwidth}
        \begin{itemize}\itemsep0.5em
            \item Write an SSM's whole sequence map as one matrix, $y = Mx$. That matrix is \textbf{semiseparable}.
            \item Apply it as a recurrence: linear time. Apply it as a masked attention matrix: quadratic time. \textbf{Same function, two algorithms.}
            \item In our notation Mamba-2 is simply
            \[ S_t=\gamma_t\,S_{t-1}+v_tk_t^\top \]
            which is \emph{gated linear attention} with a scalar gate.
            \item So ``attention versus SSM'' is a choice of \textbf{mask structure} inside one family.
        \end{itemize}
        \column{0.36\textwidth}
        \centering
        \includegraphics[width=\linewidth,height=0.6\textheight,keepaspectratio]{images/next-gen-architectures/mamba2_ssd_fig1_duality_roadmap.png}
    \end{columns}
    \src{Dao and Gu, ``Transformers are SSMs: Generalized Models and Efficient Algorithms Through Structured State Space Duality,'' \href{https://arxiv.org/abs/2405.21060}{arXiv:2405.21060}, Figure 1.}
\end{frame}

\begin{frame}{Mamba-3 (2026): Designed for Inference First}
    \begin{columns}[T,onlytextwidth]
        \column{0.5\textwidth}
        \begin{itemize}\itemsep0.25em
            \item \textbf{Better discretisation.} A trapezoidal rule replaces the separate conv layer.
            \item \textbf{Complex-valued state.} The state can \emph{rotate}, which scalar decay cannot. It acts as a data-dependent RoPE.
            \item \textbf{MIMO.} More compute per decode step, same state size.
        \end{itemize}
        \vspace{0.3em}
        \footnotesize
        \begin{adjustbox}{max width=\linewidth}\begin{tabular}{lcc}

            \bottomrule

\begin{frame}{One Family, One Table}
    Every mixer so far is a rule for updating a matrix state, read out as $o_t=S_tq_t$.
    \vspace{0.3em}
    \begin{center}
    \small
    \renewcommand{\arraystretch}{1.25}
    \begin{adjustbox}{max width=\linewidth}\begin{tabular}{lll}
        \toprule
        \textbf{Model} & \textbf{State update} & \textbf{What it adds} \\
        \midrule
        Linear attention & $S_t=S_{t-1}+v_tk_t^\top$ & nothing: write only \\
        RetNet & $S_t=\gamma\, S_{t-1}+v_tk_t^\top$ & fixed decay \\
        Mamba-2 & $S_t=\gamma_t\, S_{t-1}+v_tk_t^\top$ & input-dependent decay \\
        GLA & $S_t=S_{t-1}\mathrm{Diag}(\alpha_t)+v_tk_t^\top$ & per-channel decay \\
        DeltaNet & $S_t=S_{t-1}(I-\beta_tk_tk_t^\top)+\beta_tv_tk_t^\top$ & targeted overwrite \\
        \rowcolor{KAUSTcyan!12} Gated DeltaNet & $S_t=S_{t-1}\,\alpha_t(I-\beta_tk_tk_t^\top)+\beta_tv_tk_t^\top$ & decay and overwrite \\
        \bottomrule
    \end{tabular}\end{adjustbox}
    \end{center}
    \vspace{0.2em}
    \begin{itemize}
        \item Each transition stays \textbf{diagonal plus low-rank}. That structure is what keeps chunkwise training possible.
    \end{itemize}
    \src{After Yang et al., \href{https://arxiv.org/abs/2406.06484}{arXiv:2406.06484}, Table 2, and Yang, Kautz, Hatamizadeh, \href{https://arxiv.org/abs/2412.06464}{arXiv:2412.06464}.}
\end{frame}

\begin{frame}{What a Fixed State Cannot Do}
    \begin{columns}[T,onlytextwidth]
        \column{0.4\textwidth}
        \begin{itemize}\itemsep0.5em
            \item A fixed-size state is a \textbf{summary}. What the summary drops is gone.
            \item \textbf{Theory:} any recurrent model needs $\Omega(N)$ bits of state to solve multi-query recall over $N$ pairs.
            \item \textbf{Practice:} Transformers learn to copy from far fewer examples and generalise to longer strings.
            \item Better update rules raise the ceiling. They do not remove it.
        \end{itemize}
