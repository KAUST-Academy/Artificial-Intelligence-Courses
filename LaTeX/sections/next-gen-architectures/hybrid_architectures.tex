\section{Hybrid Architectures}

\begin{frame}{}
    \LARGE Next-Generation Architectures: \textbf{Hybrid Architectures}

    \vspace{1em}
    \normalsize If a fixed state cannot recall and attention cannot stay cheap, why not use both?
\end{frame}

\begin{frame}{A Few Attention Layers Buy Back Recall}
    \begin{itemize}\itemsep0.4em
        \item Keep the cheap recurrent mixer in most layers. Put full attention in a few.
    \end{itemize}
    \begin{center}
    \small
    \begin{adjustbox}{max width=\linewidth}\begin{tabular}{lccc}
        \toprule
        \textbf{1.3B, 100B tokens} & \textbf{Perplexity} $\downarrow$ & \textbf{LM eval} $\uparrow$ & \textbf{Retrieval} $\uparrow$ \\
        \midrule
        Transformer++ & 16.9 & 50.9 & 41.8 \\
        DeltaNet & 16.9 & 51.6 & 34.7 \\
        \rowcolor{KAUSTcyan!12} DeltaNet + attention in \textbf{2 layers} & 16.6 & 51.8 & \textbf{47.9} \\
        \bottomrule
    \end{tabular}\end{adjustbox}
    \end{center}
    \begin{itemize}\itemsep0.4em
        \item A controlled study of 72 models confirms the shape. Language-modelling quality is \textbf{flat} in the mixing ratio. Recall \textbf{rises steeply} with more attention, and saturates near 3 recurrent layers per attention layer.
        \item The KV cache shrinks by the same factor, since only attention layers keep one.
        \item Surprise from the same study: a mixer's standalone quality does not predict its quality inside a hybrid.
    \end{itemize}
    \src{Yang, ``Linear Transformers for Efficient Sequence Modeling'' (talk, 2024), slides 122 to 126. ``A Systematic Analysis of Hybrid Linear Attention,'' \href{https://arxiv.org/abs/2507.06457}{arXiv:2507.06457}.}
\end{frame}

\begin{frame}{The Hybrid Recipe in 2026}
    \begin{center}
    \small
    \renewcommand{\arraystretch}{1.2}
    \begin{adjustbox}{max width=\linewidth}\begin{tabular}{llll}
        \toprule
        \textbf{Model} & \textbf{Layer pattern} & \textbf{Total / active} & \textbf{Context} \\
        \midrule
        Qwen3.5 (Feb 2026) & 3 Gated DeltaNet : 1 attention & 397B / 17B & 262K \\
        Kimi K3 (2026) & 69 KDA + 24 attention layers & 2.8T / 104B & 1M \\
        Nemotron 3 Super (Mar 2026) & 40 Mamba-2 + 8 attention & 120.6B / 12.7B & 1M \\
        Jamba2 Mini (Jan 2026) & 7 Mamba : 1 attention & 52B / 12B & 262K \\
        Granite 4.0-H (Oct 2025) & 9 Mamba-2 : 1 Transformer & 32B / 9B & 512K \\
        Falcon-H1 (May 2025) & attention $\parallel$ Mamba-2 in every block & up to 34B & 256K \\
        \bottomrule
    \end{tabular}\end{adjustbox}
    \end{center}
    \begin{itemize}\itemsep0.4em
        \item Two camps: \textbf{delta-rule} mixers at about 3:1, and \textbf{Mamba-2} mixers with 8 to 20\% attention layers.
        \item Read the ratio, not the label. Most of these models are also sparse MoEs.
    \end{itemize}
    \src{Official model cards and configuration files (Qwen, Moonshot AI, NVIDIA, AI21, IBM, TII), 2025 to 2026. Raschka, ``LLM Architecture Gallery: Hybrid Attention'' (2026).}
\end{frame}

\begin{frame}{Kimi Linear: A Same-Recipe Test}
    \begin{itemize}\itemsep0.5em
        \item Most hybrid claims compare models trained on different data. \textbf{Kimi Linear} ran the cleanest test: same data, same recipe, against a full-attention baseline.
    \end{itemize}
    \vspace{0.3em}
    \begin{center}
    \small
    \begin{adjustbox}{max width=\linewidth}\begin{tabular}{lcc}
        \toprule
        & \textbf{Kimi Linear} & \textbf{Full attention} \\
        \midrule
        MMLU-Pro (4K) & 51.0 & 47.2 \\
        RULER (128K) & 84.3 & 81.3 \\
        \bottomrule
    \end{tabular}\end{adjustbox}
    \end{center}
    \vspace{0.3em}
    \begin{itemize}\itemsep0.5em
        \item The hybrid is ahead on short-context knowledge and on long-context retrieval at once. In this comparison the recurrent layers cost no quality.
        \item Serving had to change too. vLLM added a \textbf{hybrid cache manager} so a recurrent state and a KV block share the same memory pool.
    \end{itemize}
    \src{Kimi Team, ``Kimi Linear: An Expressive, Efficient Attention Architecture,'' \href{https://arxiv.org/abs/2510.26692}{arXiv:2510.26692}. vLLM blog, ``vLLM Now Supports Qwen3-Next'' (Sep 2025).}
\end{frame}

\begin{frame}{The Counter-Trend: Two Labs Walked Back}
    \begin{columns}[T,onlytextwidth]
        \column{0.56\textwidth}
        \textbf{MiniMax}
        \begin{itemize}\itemsep0.3em
            \item \textbf{M1} (2025): 7 linear layers per softmax layer.
            \item \textbf{M2} (Oct 2025): \alert{full attention in all 62 layers}.
            \item Their stated reasons:
            \begin{itemize}\itemsep0.2em
                \item deficits in multi-hop reasoning at scale that standard benchmarks missed
                \item higher sensitivity to numerical precision
                \item no mature prefix caching or speculative decoding for linear layers
            \end{itemize}
        \end{itemize}
        \column{0.41\textwidth}
        \textbf{IBM}
        \begin{itemize}\itemsep0.3em
            \item \textbf{Granite 4.0-H} (Oct 2025): Mamba-2 hybrid.
            \item \textbf{Granite 4.1 and 4.2} (2026): back to a dense Transformer. The 8B dense model matches or beats the larger 4.0 hybrid.
        \end{itemize}
        \vspace{0.6em}
        \caveat{Hybrids are a \textbf{contested} default, not a settled one.}
    \end{columns}
    \vspace{0.3em}
    \src{MiniMax, ``Why Did MiniMax M2 End Up as a Full Attention Model?'' (2025). IBM, ``Granite 4.1 LLMs: How They're Built'' (Apr 2026).}
\end{frame}

\begin{frame}{Where the Mixer Stands (October 2026)}
    \begin{itemize}\itemsep0.55em
        \item \textbf{Live designs:} delta-rule linear attention (Qwen, Kimi) and Mamba-2 (NVIDIA, AI21, TII). RWKV, xLSTM and Hyena keep niches.
        \item \textbf{Open problem 1, recall.} A fixed state has a capacity bound. Hybrids work around it. They do not solve it.
        \item \textbf{Open problem 2, state tracking.} Scalar decay cannot do parity or permutations. Mamba-3's rotations and RWKV-7's delta rule both attack this.
        \item \textbf{Open problem 3, systems.} Caching, quantisation and evaluation were all built for attention.
    \end{itemize}
    \vspace{0.2em}
    \takeaway{\textbf{Next.} The mixer is only half of the block. The other half, the feed-forward layer, holds most of the parameters.}
\end{frame}
