\section{Motivation}

\begin{frame}{The Transformer Was Not Replaced. It Was Re-plumbed.}
    \begin{itemize}\itemsep0.6em
        \item The block from 2017 is still recognisable in every frontier model of 2026.
        \item What changed is \textbf{where the cost goes}. Three pressures drive almost every design in this lecture:
        \begin{itemize}\itemsep0.3em
            \item \textbf{Memory per token.} Attention keeps every past token in a cache that grows with context.
            \item \textbf{Compute per token.} Most parameters sit in feed-forward layers that every token pays for.
            \item \textbf{Context length.} One million tokens is now a product feature, not a research demo.
        \end{itemize}
        \item The answers are \alert{hybrids}, \alert{sparsity} and \alert{compression}, not a new architecture.
    \end{itemize}
    \vspace{0.4em}
    \takeaway{Frontier MoE models went from about 28\% of parameters active per token (Mixtral, 2024) to about 3 to 5\% in 2026.}
    \src{Raschka, ``The Big LLM Architecture Comparison'' (2025, updated 2026). Jiang et al., ``Mixtral of Experts,'' \href{https://arxiv.org/abs/2401.04088}{arXiv:2401.04088}. Kimi K3 and DeepSeek-V4 model configurations (2026).}
\end{frame}
