\begin{frame}{A Puzzle: Better Reasoners That Hallucinate More}
    \begin{center}

        o3 (Apr 2025) & 0.49 & 0.51 & 0.00 \\
        \rowcolor{KAred!10} o4-mini (Apr 2025) & 0.20 & 0.79 & 0.01 \\

    \begin{itemize}\itemsep0.45em
        \item o3 hallucinates more than o1. OpenAI's reading: it \emph{``tends to make more claims overall''}.
        \item The two small models \textbf{know about the same amount} (0.20 and 0.22). One answers almost everything. The other declines half the time.
        \item So hallucination is not only a knowledge problem. It is a \textbf{decision}: answer, or abstain.
    \end{itemize}

\begin{frame}{Three Arguments That Some Hallucination Is Unavoidable}
    \begin{itemize}\itemsep0.3em
        \item \textbf{Terms.} A hallucination is a \emph{plausible falsehood}. A \emph{confabulation} is wrong \emph{and arbitrary}: it changes with the random seed.
    \end{itemize}

\begin{frame}{The Incentive: Under Binary Grading, Guessing Is Optimal}
    \begin{itemize}\itemsep0.45em
        \item \textbf{Observation.} If a grader gives 1 for a right answer and 0 otherwise, ``I don't know'' is \emph{never} the best response. A student facing such an exam guesses.
        \item \textbf{Audit.} GPQA, MMLU-Pro, IFEval, BBH, MATH, SWE-bench and HLE give no credit for abstaining. Models are tuned on these scoreboards.
        \item \textbf{Proposed fix.} State a confidence target in the task, and score accordingly:
    \end{itemize}
    \begin{center}
    \begin{adjustbox}{max width=\linewidth}\begin{tikzpicture}[font=\small]
        \node[draw, rounded corners=2pt, fill=gray!10, text width=12.4cm, align=left, inner sep=5pt] at (0,0) {\emph{``Answer only if you are $> t$ confident, since mistakes are penalized $t/(1-t)$ points, while correct answers receive 1 point.''}};
    \end{tikzpicture}\end{adjustbox}
    \end{center}
    \begin{itemize}\itemsep0.45em
        \item The authors ask for this in \emph{mainstream} benchmarks, not in one more hallucination benchmark.
        \item \textbf{Not settled.} A formal result says wrong definite answers cannot be removed entirely, so the real quantity is \emph{coverage} at guaranteed correctness.
    \end{itemize}
    \src{Kalai, Nachum, Vempala, Zhang, ``Why Language Models Hallucinate,'' \href{https://arxiv.org/abs/2509.04664}{arXiv:2509.04664}, Observation 1. Imamura, ``Hallucination, abstention, and computable inseparability,'' \href{https://arxiv.org/abs/2604.28067}{arXiv:2604.28067}.}
\end{frame}

\begin{frame}{Semantic Entropy: Uncertainty over Meanings, Not Strings}
    \begin{columns}[T,onlytextwidth]
        \column{0.58\textwidth}
        \begin{itemize}\itemsep0.4em
            \item \textbf{Problem.} Two paraphrases of one answer are two strings. Token entropy sees disagreement.
            \item \textbf{Method.} Sample several answers. Put two in the same cluster $C$ if each \emph{entails} the other. Take the entropy over clusters:
            \[
                \mathrm{SE}(x)\approx-\sum_{i}P(C_i\mid x)\,\log P(C_i\mid x)
            \]
            \item A \emph{discrete} variant only counts samples per cluster, so it runs on closed models.
            \item \textbf{Limit.} It detects \emph{arbitrary} errors. A model that is consistently wrong has low entropy.
        \end{itemize}
        \column{0.4\textwidth}
        \centering

\begin{frame}{Acting on Uncertainty: Guarantee, Abstain, Verify}
    \begin{center}
    \small
    \renewcommand{\arraystretch}{1.15}
    \begin{adjustbox}{max width=\linewidth}\begin{tabular}{>{\raggedright\arraybackslash}p{2.9cm}>{\raggedright\arraybackslash}p{5.3cm}>{\raggedright\arraybackslash}p{4.4cm}}
        \toprule
        \textbf{Approach} & \textbf{How} & \textbf{Reported effect} \\
        \midrule
        Conformal factuality & drop the least supported sub-claims until a calibrated threshold is met & 80 to 90\% correctness guarantees \\
        R-Tuning & fine-tune on data labelled with what the model does not know & refuses 87 to 99\% of unanswerable questions \\
        Chain-of-Verification & draft, write checking questions, answer them \emph{without} the draft, revise & FActScore 55.9 to 71.4 \\
        Abstention-aware RL & reward declining when likely wrong & up to 10.3 points of precision (one 2026 preprint) \\
        \bottomrule
    \end{tabular}\end{adjustbox}
    \end{center}
    \begin{itemize}\itemsep0.4em
        \item The trend follows the diagnosis: from decoding tricks (2023) to \textbf{rewards that pay for abstention} (2026).
        \item \textbf{The open trade-off:} precision against coverage. Abstaining on everything is trivially safe.
    \end{itemize}
    \src{Mohri, Hashimoto, \href{https://arxiv.org/abs/2402.10978}{arXiv:2402.10978}. Zhang, Diao et al., ``R-Tuning,'' \href{https://arxiv.org/abs/2311.09677}{arXiv:2311.09677}, Table 3. \begin{frame}{Hallucination: What to Remember}
    \begin{itemize}\itemsep0.55em
        \item Some hallucination is \textbf{statistically unavoidable} on facts seen once. Much of the rest is \textbf{an incentive}: binary grading rewards a guess over ``I don't know''.
        \item \textbf{Post-training makes it worse}: RLHF degrades calibration, and reasoning fine-tuning lowers abstention.
        \item \textbf{Uncertainty estimators} trade cost against access. Semantic entropy catches arbitrary errors and misses systematic ones.
        \item The fixes that match the diagnosis are \textbf{abstention-aware scoring and rewards}.
        \item Read a hallucination rate \textbf{together with accuracy and abstention}.
    \end{itemize}
    \vspace{0.4em}
    \takeaway{\textbf{Next.} Some 2026 preprints suggest the model \emph{has} the uncertainty signal inside and does not act on it. To check a claim like that, we have to look inside.}
\end{frame}
