\section{Motivation}

\begin{frame}{The Frontier Is No Longer a Bigger Dense Model}
    \begin{itemize}\itemsep0.55em
        \item Days 1 to 4 followed one model from architecture to reasoning, post-training and agents. Today asks: \textbf{what is still moving?}
        \item A 2026 phone-sized model, Gemma 4 E2B, \emph{``roughly matches Gemma 3 27B performance with 10x less parameters''}.
        \item So the research questions changed. Three of them organise this lecture:
        \begin{itemize}\itemsep0.25em
            \item \textbf{Where does computation live?} On the edge, in a drafter, in a parallel denoiser.
            \item \textbf{Where does memory live?} In dense weights, in addressable tables, in the context.
            \item \textbf{How is any of it checked?} By uncertainty estimates, by looking inside, by judges.
        \end{itemize}
    \end{itemize}
    \vspace{0.3em}
    \takeaway{Ask for the honest number. Speculative decoding with EAGLE-3 gives 4.40x at batch size 1, and 1.38x at batch size 64, on the same model.}
    \src{Gemma Team, ``Gemma 4 Technical Report'' (2026), \href{https://arxiv.org/abs/2607.02770}{arXiv:2607.02770}. Li et al., ``EAGLE-3,'' \href{https://arxiv.org/abs/2503.01840}{arXiv:2503.01840}, Tables 1 and 3 (LLaMA-3.1-8B).}
\end{frame}
