\section{Motivation}

\begin{frame}{The Agent Is No Longer a Prompt}
    \begin{itemize}\itemsep0.45em
        \item In 2023 an agent was a \emph{prompt scaffold} wrapped around a chat model. ReWOO, LATS, multi-agent debate and Toolformer are now history lessons.
        \item By 2026, agency is \textbf{trained into the model} and \textbf{engineered around it}:
        \begin{itemize}\itemsep0.3em
            \item \textbf{Trained behaviour.} Kimi K2 learns from 3,000+ real MCP tools and 20,000+ synthetic ones. DeepSeek-V3.2 trains in 1,800+ environments.
            \item \textbf{Planning inside the model.} Every major API now asks the client to pass reasoning state back across tool calls.
            \item \textbf{An engineered harness.} Context, memory, permissions and sandboxes are designed, not prompted.
        \end{itemize}
    \end{itemize}
    \vspace{0.3em}
    \takeaway{Public frontier agents finish tasks of about \textbf{12 hours} at 50\% success, but only about \textbf{1.5 hours} at 80\%. Capability is ahead of reliability.}
    \src{Kimi Team, ``Kimi K2: Open Agentic Intelligence,'' \href{https://arxiv.org/abs/2507.20534}{arXiv:2507.20534}. DeepSeek-AI, ``DeepSeek-V3.2,'' \href{https://arxiv.org/abs/2512.02556}{arXiv:2512.02556}. METR, ``Frontier Risk Report'' (May 2026).}
\end{frame}
