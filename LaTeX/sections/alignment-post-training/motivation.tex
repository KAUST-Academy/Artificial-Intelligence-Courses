\section{Motivation}

\begin{frame}{Post-Training Moved. The Bottleneck Moved with It.}
    \begin{itemize}\itemsep0.35em
        \item In 2023 and 2024 the centre of post-training was \textbf{RLHF and DPO} on human preferences.
        \item By 2025 and 2026 the open recipe is SFT, then an optional DPO-style stage, then \alert{large-scale on-policy RL with verifiable rewards}.
        \item The scarce resource is no longer the optimiser. It is the \textbf{signal}:
        \begin{itemize}\itemsep0.3em
            \item \textbf{Cost.} A human preference label costs 5 to 20 dollars. An AI label costs under one cent.
            \item \textbf{Data.} More than 98\% of the alignment data of Nemotron-4 340B is synthetic.
            \item \textbf{Compute.} DeepSeek-V3.2 reports post-training compute above 10\% of its pre-training cost.
        \end{itemize}
        \item Once humans leave the loop, who checks the reward, the data and the model?
    \end{itemize}
    \vspace{0.3em}
    \takeaway{Every oversight signal degrades when you optimise against it: a reward model, a judge, a chain-of-thought monitor, a self-generated label.}
    \src{Lambert, ``The state of post-training in 2025'' (Interconnects, 2025). NVIDIA, ``Nemotron-4 340B Technical Report,'' \href{https://arxiv.org/abs/2406.11704}{arXiv:2406.11704}. DeepSeek-AI, ``DeepSeek-V3.2,'' \href{https://arxiv.org/abs/2512.02556}{arXiv:2512.02556}.}
\end{frame}
