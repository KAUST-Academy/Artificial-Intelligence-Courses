\section{Continual Post-Training and Model Merging}

\begin{frame}{}
    \LARGE Alignment and Post-Training: \textbf{Continual Post-Training and Merging}

    \vspace{1em}
    \normalsize A deployed model is never finished. How do you add to it without breaking what it knew?
\end{frame}

\begin{frame}{Recap, and Three Axes for the Rest}
    \begin{columns}[T,onlytextwidth]
        \column{0.5\textwidth}
        \textbf{Recap: four defences against forgetting}
        \begin{itemize}\itemsep0.4em
            \item \textbf{Replay.} Mix a slice of the old data into every new run.
            \item \textbf{Regularise.} Penalise movement along directions that mattered, as in elastic weight consolidation.
            \item \textbf{Constrain.} LoRA learns less than full fine-tuning, and forgets less.
            \item \textbf{Merge.} Train the new skill separately, then combine weights.
        \end{itemize}
        \column{0.47\textwidth}
        \textbf{A 2026 map of continual learning}
        \begin{center}
        \small
        \renewcommand{\arraystretch}{1.25}
        \begin{adjustbox}{max width=\linewidth}\begin{tabular}{l>{\raggedright\arraybackslash}p{4.3cm}}
            \toprule
            \textbf{When} & pre-training, post-training, inference time \\
            \textbf{How} & off-policy, on-policy, beyond gradients \\
            \textbf{Where} & parameters, or memory, skills and protocols \\
            \bottomrule
        \end{tabular}\end{adjustbox}
        \end{center}
    \end{columns}
    \vspace{0.6em}
    \begin{itemize}\itemsep0.4em
        \item The survey's thesis: adaptation is moving from the \textbf{parameters} to the \textbf{system} around them.
    \end{itemize}

\begin{frame}{Continual Pre-Training: Re-Warm, Re-Decay, Replay}
    \begin{columns}[T,onlytextwidth]
        \column{0.55\textwidth}
        \begin{itemize}\itemsep0.45em
            \item \textbf{Claim.} Re-warm the learning rate, decay it again, and replay a little old data. The result matches re-training on the union of both corpora, at a fraction of the compute.
            \item Final validation loss at 10B parameters: \textbf{1.89} continual against \textbf{1.87} re-trained.
            \item \textbf{How much replay?} 5\% for a weak shift, 25\% for a strong one. Even 0.5\% sharply reduces forgetting.
            \item \textbf{Two sources of forgetting.} Re-warming causes a loss spike \emph{even on the same data}. That part is optimisation, not distribution shift.
        \end{itemize}
        \column{0.43\textwidth}
        \centering
        \small
        \begin{adjustbox}{max width=\linewidth}\begin{tabular}{lr}
            \toprule
            \textbf{Recipe} & \textbf{Replay share} \\
            \midrule
            Ibrahim et al., weak shift & 5\% \\
            Ibrahim et al., strong shift & 25\% \\
            Code Llama & 7\% \\
            Llemma & 5\% \\
            AdaptLLM (instructions) & 50 to 67\% \\
            \bottomrule
        \end{tabular}\end{adjustbox}

        \vspace{0.5em}
        {\scriptsize Share of general-domain data kept in the continued run.}
    \end{columns}
    \vspace{0.3em}
    \src{Ibrahim et al., ``Simple and Scalable Strategies to Continually Pre-train Large Language Models,'' \href{https://arxiv.org/abs/2403.08763}{arXiv:2403.08763}, Table 4 and Figure 8. Replay shares from the Code Llama, Llemma and AdaptLLM papers (cited on the next slide).}
\end{frame}

\begin{frame}{Fine-Tuning Forgets by Law, and Safety Goes First}
    \begin{columns}[T,onlytextwidth]
        \column{0.5\textwidth}
        \textbf{A scaling law for forgetting}
        \[
            \mathcal{L}_{f}=-c\,\mathcal{L}_{ft}+s
        \]
        \begin{itemize}\itemsep0.4em
            \item Forgetting $\mathcal{L}_f$ is \emph{linear} in the fine-tuning loss, with $R^2$ of 0.95 and 0.97 on two datasets.
            \item Neither early stopping nor a different LoRA rank escapes the line.
            \item For RL, LoRA matches full fine-tuning even at rank 1. Policy gradient absorbs only about $O(1)$ bits per episode.
        \end{itemize}
        \column{0.47\textwidth}
        \textbf{Safety is the most fragile behaviour}
        \begin{center}
        \small
        \begin{adjustbox}{max width=\linewidth}\begin{tabular}{lr}
            \toprule
            \textbf{GPT-3.5 Turbo, fine-tuned on} & \textbf{Harmful} \\
            \midrule
            nothing (baseline) & 1.8\% \\
            10 harmful examples & 88.8\% \\
            benign Alpaca data & 31.8\% \\
            Alpaca plus 250 safety samples & 21.8\% \\
            \bottomrule
        \end{tabular}\end{adjustbox}
        \end{center}
        \begin{itemize}\itemsep0.4em
            \item The 10 examples cost under 0.20 dollars.
            \item Benign data alone erodes safety. Mixing safety data back restores it only partly.
        \end{itemize}
    \end{columns}
    \vspace{0.3em}
    \src{Kalajdzievski, ``Scaling Laws for Forgetting When Fine-Tuning Large Language Models,'' \href{https://arxiv.org/abs/2401.05605}{arXiv:2401.05605}, Eq. 4 and Figure 2. Schulman, ``LoRA Without Regret'' (Thinking Machines, 2025). Qi et al., ``Fine-tuning Aligned Language Models Compromises Safety,'' \href{https://arxiv.org/abs/2310.03693}{arXiv:2310.03693}.}
\end{frame}

\begin{frame}{SFT or RL: Which Forgets Less?}
    \centering
    \includegraphics[width=0.74\linewidth,height=0.26\textheight,keepaspectratio]{images/alignment-post-training/rlrazor_fig1_teaser.png}
    \begin{columns}[T,onlytextwidth]
        \column{0.36\textwidth}
        \centering
        \small
        \begin{adjustbox}{max width=\linewidth}\begin{tabular}{lcc}
            \toprule
            \textbf{Llama-3.1-8B} & \textbf{Target} & \textbf{Other} \\
            \midrule
            SFT & $+25.2$ & $-27.8$ \\
            \rowcolor{KAUSTcyan!12} RL & $+18.4$ & $-3.4$ \\
            \bottomrule
        \end{tabular}\end{adjustbox}
        \column{0.62\textwidth}
        \begin{itemize}\itemsep0.3em
            \item \textbf{RL's Razor.} RL finds the solution closest in KL to the base model. Forgetting tracks $\mathbb{E}_{x}\big[\mathrm{KL}(\pi_0\,\|\,\pi)\big]$ on the \emph{new} task's inputs.
            \item \textbf{The debate.} A second study finds a weaker link to KL (correlation 0.52) and credits \emph{on-policy data} instead.
        \end{itemize}
    \end{columns}
    \vspace{0.2em}
    \caveat{This holds for a \textbf{single} new task. In sequential multi-task RL, the drift happens on old tasks and RL does forget (2026).}
    \src{Shenfeld, Pari, Agrawal, ``RL's Razor: Why Online RL Forgets Less,'' \href{https://arxiv.org/abs/2509.04259}{arXiv:2509.04259}, Figure 1. Chen et al., ``Retaining by Doing,'' \href{https://arxiv.org/abs/2510.18874}{arXiv:2510.18874}. Luo et al., ``RL Forgets! Towards Continual Policy Optimization,'' \href{https://arxiv.org/abs/2607.04364}{arXiv:2607.04364}.}
\end{frame}

\begin{frame}{Model Merging in One Slide (Recap)}
    \centering
    \includegraphics[width=0.8\linewidth,height=0.26\textheight,keepaspectratio]{images/alignment-post-training/task_arithmetic_fig1.png}
    \begin{itemize}\itemsep0.3em
        \item \textbf{Task vector.} What a fine-tune learned is a direction in weight space: $\tau=\theta_{\text{ft}}-\theta_{\text{pre}}$. Negate it, add it, or combine several: $\theta_{\text{new}}=\theta_{\text{pre}}+\lambda\sum_i\tau_i$.
        \item \textbf{Recipes.} Linear averaging (soup), SLERP, TIES (trim, elect sign, merge) and DARE (drop and rescale).
        \item \textbf{Two rules.} Merge only models that share a pre-trained checkpoint. Evaluate the merge itself.
    \end{itemize}
    \vspace{0.1em}
    \takeaway{Next: where merging sits inside the pipelines of 2024 to 2026.}
    \src{Ilharco et al., ``Editing Models with Task Arithmetic,'' ICLR 2023, \href{https://arxiv.org/abs/2212.04089}{arXiv:2212.04089}, Figure 1. Yadav et al., ``TIES-Merging,'' \href{https://arxiv.org/abs/2306.01708}{arXiv:2306.01708}. Recap of the KAUST Academy LLM course.}
\end{frame}

\begin{frame}{Merging Inside Real Recipes}
    \begin{itemize}\itemsep0.3em
        \item \textbf{Against the alignment tax.} RLHF raises the reward while reading comprehension and translation scores fall. Interpolating back toward the base model gave the best trade-off among all methods compared, including replay, LoRA and a KL penalty:
    \end{itemize}
    \[
        \theta=(1-\alpha)\,\theta_0+\alpha\,\theta_{\text{RLHF}}, \qquad \alpha=0.2 \text{ by default}
    \]
    \begin{center}
    \small
    \begin{adjustbox}{max width=\linewidth}\begin{tabular}{l>{\raggedright\arraybackslash}p{10.8cm}}
        \toprule
        \textbf{Recipe} & \textbf{How merging is used} \\
        \midrule
        OLMo 2 & average several mid-training runs that differ only in data order \\
        T\"ulu 3 & linear averaging with mergekit \\
        SmolLM3 & merge the aligned model 0.9 to 0.1 with the mid-training checkpoint, to recover long-context performance at 128k \\
        \bottomrule
    \end{tabular}\end{adjustbox}
    \end{center}
    \begin{itemize}\itemsep0.3em
        \item \textbf{A 2026 limit.} For combining several RL specialists, multi-teacher on-policy distillation beat weight merging.
    \end{itemize}

\begin{frame}{Continual Post-Training: What to Remember}
    \begin{itemize}\itemsep0.6em
        \item Continual pre-training works with three ingredients: \textbf{re-warm, re-decay, replay}. A few per cent of replay goes a long way.
        \item Domain adaptation has a bill. Check \textbf{off-domain skills} and \textbf{prompting ability}, not only the target benchmark.
        \item Forgetting is linear in how much you learn. \textbf{Safety behaviour erodes first}, even under benign fine-tuning.
        \item On-policy updates forget less for a single task. Sequential RL still forgets.
        \item Merging is now a routine pipeline step: against the alignment tax, across data orders, and to recover a lost skill.
    \end{itemize}
